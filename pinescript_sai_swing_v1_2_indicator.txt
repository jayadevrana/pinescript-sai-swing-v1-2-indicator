eSignal = 0
eBuy = close > NewH and (nz(eSignal[1]) == 0 or nz(eSignal[1]) == -1)
eShort = close < NewL and (nz(eSignal[1]) == 0 or nz(eSignal[1]) == 1)
eSell = close < NewH and nz(eSignal[1]) == 1
eCover = close > NewL and nz(eSignal[1]) == -1
eSignal := eBuy ? 1 : eShort ? -1 : eSell or eCover ? 0 : eSignal[1]

eSellLow = ta.valuewhen(eSell, low, 0)
eCoverHigh = ta.valuewhen(eCover, high, 0)

eBuyBars = ta.barssince(eBuy)
eShortBars = ta.barssince(eShort)
eSellBars = ta.barssince(eSell)
eCoverBars = ta.barssince(eCover)

BuySignal = eSignal[1] > 0
ShortSignal = eSignal[1] < 0
SellSignal = low < eSellLow and eSellBars < eBuyBars
CoverSignal = high > eCoverHigh and eCoverBars < eShortBars

CoverFinal = CoverSignal and nz(MainSignal[1]) == -1
MainSignal := BuySignal ? 1 : ShortSignal ? -1 : SellSignal ? 0 : CoverSignal ? 0 : MainSignal[1]
